cmake_minimum_required(VERSION 3.20)
project(v80_opentpu LANGUAGES CXX)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(generated "${CMAKE_CURRENT_BINARY_DIR}/generated")
foreach(script configure generate_contracts)
  execute_process(COMMAND "${CMAKE_COMMAND}" -E env
    "PYTHONPATH=${CMAKE_SOURCE_DIR}:${CMAKE_SOURCE_DIR}/third_party/openTPU"
    "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/software/tools/${script}.py" --output "${generated}"
    COMMAND_ERROR_IS_FATAL ANY)
endforeach()
add_library(v80_driver STATIC software/driver/backends/qdma/v80_qdma.cpp software/runtime/v80_runtime.cpp)
target_compile_features(v80_driver PUBLIC cxx_std_20)
target_compile_options(v80_driver PRIVATE -Wall -Wextra -Werror)
target_include_directories(v80_driver PUBLIC software/driver/include software/runtime "${generated}")
add_executable(v80-info software/driver/tools/v80_info.cpp)
target_link_libraries(v80-info PRIVATE v80_driver)
add_executable(v80-smoke software/driver/tools/v80_smoke.cpp)
target_compile_options(v80-smoke PRIVATE -Wall -Wextra -Werror)
target_link_libraries(v80-smoke PRIVATE v80_driver)
enable_testing()
add_executable(v80_driver_test tests/unit/driver_test.cpp)
target_include_directories(v80_driver_test PRIVATE software/driver/backends/mock)
target_compile_options(v80_driver_test PRIVATE -Wall -Wextra -Werror)
target_link_libraries(v80_driver_test PRIVATE v80_driver)
add_test(NAME v80_driver_test COMMAND v80_driver_test)
